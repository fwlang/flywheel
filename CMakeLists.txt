cmake_minimum_required(VERSION 3.24)
project(Flywheel VERSION 0.1.0 LANGUAGES C CXX)
# Pre-release tag of this version; empty for a final release. `flywheel --version` reports both.
set(FLYWHEEL_PRERELEASE "beta.1")
if(FLYWHEEL_PRERELEASE)
  set(FLYWHEEL_VERSION "${PROJECT_VERSION}-${FLYWHEEL_PRERELEASE}")
else()
  set(FLYWHEEL_VERSION "${PROJECT_VERSION}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
if(LLVM_VERSION_MAJOR LESS 21 OR LLVM_VERSION_MAJOR GREATER 23)
  message(FATAL_ERROR "Flywheel requires LLVM 21 through 23")
endif()
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION}")
add_library(flywheel_runtime STATIC runtime/runtime.cpp)
target_link_libraries(flywheel_runtime PUBLIC ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(flywheel_runtime PUBLIC ws2_32)
endif()
set_target_properties(flywheel_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(flywheel src/frontend.cpp src/modules.cpp src/compiler.cpp src/format.cpp src/lsp.cpp src/main.cpp)
target_include_directories(flywheel PRIVATE include)
target_include_directories(flywheel SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITION_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_options(flywheel PRIVATE ${LLVM_DEFINITION_LIST})
llvm_map_components_to_libnames(FLYWHEEL_LLVM_LIBS core support native passes target mc codegen orcjit)
if(TARGET LLVM)
  target_link_libraries(flywheel PRIVATE LLVM)
else()
  target_link_libraries(flywheel PRIVATE ${FLYWHEEL_LLVM_LIBS})
endif()
file(TO_CMAKE_PATH "${CMAKE_CXX_COMPILER}" FW_CXX_PATH)
# `flywheel build` links through FW_CXX; mold and lld link programs several times faster than GNU ld.
set(FLYWHEEL_LINKER "auto" CACHE STRING "Linker for flywheel build: auto, mold, lld, or default")
set(FW_LINKER_FLAG "")
if(NOT MSVC AND NOT FLYWHEEL_LINKER STREQUAL "default")
  include(CheckLinkerFlag)
  set(FW_LINKER_CANDIDATES ${FLYWHEEL_LINKER})
  if(FLYWHEEL_LINKER STREQUAL "auto")
    set(FW_LINKER_CANDIDATES mold lld)
  endif()
  foreach(linker IN LISTS FW_LINKER_CANDIDATES)
    check_linker_flag(CXX "-fuse-ld=${linker}" FW_HAS_LINKER_${linker})
    if(FW_HAS_LINKER_${linker})
      set(FW_LINKER_FLAG "-fuse-ld=${linker}")
      break()
    endif()
  endforeach()
  if(NOT FLYWHEEL_LINKER STREQUAL "auto" AND NOT FW_LINKER_FLAG)
    message(FATAL_ERROR "FLYWHEEL_LINKER=${FLYWHEEL_LINKER} is not supported by ${CMAKE_CXX_COMPILER}")
  endif()
endif()
message(STATUS "flywheel build links with: ${FW_CXX_PATH} ${FW_LINKER_FLAG}")
target_compile_definitions(flywheel PRIVATE FW_VERSION="${FLYWHEEL_VERSION}" FW_CXX="${FW_CXX_PATH}" FW_LINKER_FLAG="${FW_LINKER_FLAG}" FW_RUNTIME="$<TARGET_FILE:flywheel_runtime>" FW_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
add_dependencies(flywheel flywheel_runtime)
# `flywheel run` JIT-loads the program and resolves runtime calls against a copy of the runtime
# linked into the compiler. Other platforms link a temporary executable instead.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_SYSTEM_NAME MATCHES "BSD")
  target_link_libraries(flywheel PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,flywheel_runtime>")
  target_link_options(flywheel PRIVATE "LINKER:--export-dynamic-symbol=fw_*")
  target_compile_definitions(flywheel PRIVATE FW_JIT)
endif()
if(MSVC)
  target_compile_options(flywheel PRIVATE /W4 /EHsc)
else()
  target_compile_options(flywheel PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compiler.py $<TARGET_FILE:flywheel>)
install(TARGETS flywheel flywheel_runtime RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
include(cmake/Desktop.cmake)
